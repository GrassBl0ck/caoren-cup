cmake_minimum_required(VERSION 3.16)
project(CaorenCupEngine LANGUAGES CXX)
add_library(caorencup_engine SHARED native.cpp)
target_compile_features(caorencup_engine PRIVATE cxx_std_17)
target_compile_options(caorencup_engine PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_libraries(caorencup_engine PRIVATE dl)
include(CTest)
if(BUILD_TESTING)
  add_executable(caorencup_engine_test native-test.cpp)
  target_compile_features(caorencup_engine_test PRIVATE cxx_std_17)
  target_compile_options(caorencup_engine_test PRIVATE -UNDEBUG)
  target_link_libraries(caorencup_engine_test PRIVATE caorencup_engine)
  add_test(NAME native_state_restoration COMMAND caorencup_engine_test)
endif()
